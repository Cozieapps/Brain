# Desktop build of the on-device engine plus test_cli, for checking against real models.
cmake_minimum_required(VERSION 3.22)
project(beside_native_test CXX C)
set(CMAKE_CXX_STANDARD 17)
set(LLAMA_SRC "" CACHE PATH "llama.cpp source checkout")
set(LLAMA_BUILD_COMMON ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_APP OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
add_subdirectory(${LLAMA_SRC} llama)
set(SRC ${CMAKE_CURRENT_LIST_DIR}/../app/src/main/cpp)
add_executable(beside-test ${SRC}/engine.cpp ${SRC}/chat_format.cpp ${SRC}/test_cli.cpp)
add_executable(beside-format-test ${SRC}/chat_format.cpp ${SRC}/test_format.cpp)
target_link_libraries(beside-format-test PRIVATE llama-common llama)
target_link_libraries(beside-test PRIVATE llama-common llama)
